Script SQL Supabase (Table feedbacks)

Pour créer directement la table dans votre projet Supabase, ouvrez le SQL Editor de Supabase et exécutez ce script : 

-- Création de la table feedbacks pour Dabali Game
CREATE TABLE IF NOT EXISTS public.feedbacks (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    player_name TEXT NOT NULL DEFAULT 'Joueur',
    rating INTEGER NOT NULL CHECK (rating >= 1 AND rating <= 5),
    category TEXT NOT NULL CHECK (category IN ('avis', 'suggestion', 'bug')),
    message TEXT NOT NULL,
    level INTEGER DEFAULT 1,
    status TEXT NOT NULL DEFAULT 'nouveau',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Activation de la sécurité Row Level Security (RLS)
ALTER TABLE public.feedbacks ENABLE ROW LEVEL SECURITY;

-- Autoriser tout joueur à soumettre un avis (INSERT public)
CREATE POLICY "Public insert feedbacks" ON public.feedbacks
    FOR INSERT TO anon, authenticated
    WITH CHECK (true);

-- Autoriser la lecture des avis (SELECT)
CREATE POLICY "Public select feedbacks" ON public.feedbacks
    FOR SELECT TO anon, authenticated
    USING (true);

-- Autoriser la mise à jour (UPDATE status)
CREATE POLICY "Public update feedbacks" ON public.feedbacks
    FOR UPDATE TO anon, authenticated
    USING (true);
